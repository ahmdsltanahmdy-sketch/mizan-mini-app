function doGet(e) { var category = e.parameter.category || 'home'; var cache = CacheService.getScriptCache(); var cacheKey = 'mizan_news_v8_' + category; var cachedData = cache.get(cacheKey); if (cachedData != null) { var output = ContentService.createTextOutput(cachedData); output.setMimeType(ContentService.MimeType.JSON); return output; } var feeds = { home: 'https://www.mizanonline.ir/fa/rss/allnews', all: 'https://www.mizanonline.ir/fa/rss/allnews', judicial: 'https://www.mizanonline.ir/fa/rss/17', humanRights: 'https://www.mizanonline.ir/fa/rss/7', legalMag: 'https://www.mizanonline.ir/fa/rss/16', politics: 'https://www.mizanonline.ir/fa/rss/10', society: 'https://www.mizanonline.ir/fa/rss/9', economy: 'https://www.mizanonline.ir/fa/rss/11', culture: 'https://www.mizanonline.ir/fa/rss/12', sports: 'https://www.mizanonline.ir/fa/rss/13', world: 'https://www.mizanonline.ir/fa/rss/14', multimedia: 'https://www.mizanonline.ir/fa/rss/2', photos: 'https://www.mizanonline.ir/fa/rss/4', infographic: 'https://www.mizanonline.ir/fa/rss/3' }; var url = feeds[category] || feeds.home; try { var xml = UrlFetchApp.fetch(url, {muteHttpExceptions: true}).getContentText(); var items = extractRssItems(xml); var jsonString = JSON.stringify({success: true, items: items}); cache.put(cacheKey, jsonString, 300); var output = ContentService.createTextOutput(jsonString); output.setMimeType(ContentService.MimeType.JSON); return output; } catch(err) { var errorOutput = ContentService.createTextOutput(JSON.stringify({success: false, error: err.toString()})); errorOutput.setMimeType(ContentService.MimeType.JSON); return errorOutput; } } function extractRssItems(xml) { var items = []; var itemMatches = xml.match(/([\s\S]*?)<\/item>/g); if (!itemMatches) return items; for (var i = 0; i < itemMatches.length; i++) { var itemXml = itemMatches[i]; var title = extractTag(itemXml, 'title'); var link = extractTag(itemXml, 'link'); var pubDate = extractTag(itemXml, 'pubDate'); var desc = extractTag(itemXml, 'description'); var imageUrl = ''; var imgMatch = desc.match(/src="([^"]+)"/i) || desc.match(/src='([^']+)'/i); if (imgMatch && imgMatch[1]) { imageUrl = imgMatch[1]; } else { var encMatch = itemXml.match(/enclosure[^>]+url="([^"]+)"/i) || itemXml.match(/url="([^"]+)"/i); if (encMatch && encMatch[1]) { imageUrl = encMatch[1]; } } var cleanDesc = desc.replace(/<[^>]*>?/gm, '').trim(); items.push({ title: title, link: link, pubDate: pubDate, contentSnippet: cleanDesc.substring(0, 150), fullContent: cleanDesc, imageUrl: imageUrl }); } return items; } function extractTag(xml, tag) { var match = xml.match(new RegExp('<' + tag + '[^>]*>([\\s\\S]*?)<\/' + tag + '>', 'i')); return match ? match[1].replace(//g, '$1').trim() : ''; }
